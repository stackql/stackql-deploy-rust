1//! Fatal error detection for StackQL query execution.
2//!
3//! Maintains a list of error patterns that indicate unrecoverable failures
4//! (network issues, auth failures, etc.) vs normal operational errors
5//! (404 not found) that the retry/statecheck logic can handle.
6
7/// Error patterns that indicate a fatal, non-retryable failure.
8///
9/// These are checked against the error message string returned by the
10/// StackQL engine. If any pattern matches, the operation is aborted
11/// immediately rather than retried.
12///
13/// Three categories:
14///
15/// 1. **Network errors** - The request never reached the API. Any result
16///    from a query in this state is untrustworthy (e.g., an exists check
17///    returning empty could cause a duplicate resource to be created).
18///
19/// 2. **HTTP status errors** - The request reached the API but the response
20///    indicates an unrecoverable problem (auth failure, forbidden, etc.).
21///    404 is explicitly excluded as it's normal for exists checks.
22///
23/// 3. **Planner/compiler errors** - The stackql engine could not plan the
24///    query (unknown column/symbol, no matching provider operation, SQL
25///    syntax error). These are deterministic: the same query will fail the
26///    same way on every attempt, so retrying only burns the retry budget.
27const FATAL_ERROR_PATTERNS: &[&str] = &[
28    // Network-layer errors (Go net/http)
29    "dial tcp:",
30    "Client.Timeout exceeded",
31    "connection refused",
32    "no such host",
33    "request canceled while waiting for connection",
34    "request canceled (Client.Timeout",
35    "tls: handshake",
36    "certificate",
37    "network is unreachable",
38    "connection reset by peer",
39    "broken pipe",
40    "EOF",
41    // HTTP status codes that are never retryable
42    "http response status code: 401",
43    "http response status code: 403",
44    // stackql planner/compiler errors - deterministic, never succeed on retry
45    "could not locate symbol",
46    "cannot find matching operation",
47    "disparity in fields to insert",
48    "syntax error at position",
49];
50
51/// Patterns that indicate a non-fatal error, even if a fatal pattern
52/// also matches. These take precedence over `FATAL_ERROR_PATTERNS`.
53///
54/// For example, a 404 is normal for exists checks on resources that
55/// don't exist yet.
56const NON_FATAL_OVERRIDES: &[&str] = &[
57    "http response status code: 404",
58    "ResourceNotFoundException",
59    "was not found",
60];
61
62/// Check if an error message indicates a fatal, non-retryable failure.
63///
64/// Returns `Some(reason)` if the error is fatal, `None` if it's
65/// a normal operational error that can be retried or handled.
66pub fn check_fatal_error(error_msg: &str) -> Option<&'static str> {
67    // First check if any non-fatal override matches
68    for pattern in NON_FATAL_OVERRIDES {
69        if error_msg.contains(pattern) {
70            return None;
71        }
72    }
73
74    // Then check for fatal patterns
75    FATAL_ERROR_PATTERNS
76        .iter()
77        .find(|&&pattern| error_msg.contains(pattern))
78        .copied()
79}

81#[cfg(test)]
82mod tests {
83    use super::*;
84
85    #[test]
86    fn test_network_timeout_is_fatal() {
87        let msg = r#"Query execution failed: query returns error: Post "https://cloudcontrolapi.us-east-1.amazonaws.com/?Action=GetResource&Version=2021-09-30": net/http: request canceled while waiting for connection (Client.Timeout exceeded while awaiting headers)"#;
88        assert!(check_fatal_error(msg).is_some());
89    }
90
91    #[test]
92    fn test_dns_failure_is_fatal() {
93        let msg = r#"Query execution failed: query returns error: Post "https://cloudcontrolapi.us-east-1.amazonaws.com/": dial tcp: lookup cloudcontrolapi.us-east-1.amazonaws.com on 8.8.4.4:53: i/o timeout"#;
94        assert!(check_fatal_error(msg).is_some());
95    }
96
97    #[test]
98    fn test_403_is_fatal() {
99        let msg = r#"http response status code: 403, response body: {"message":"Access Denied"}"#;
100        assert!(check_fatal_error(msg).is_some());
101    }
102
103    #[test]
104    fn test_401_is_fatal() {
105        let msg = r#"http response status code: 401, response body: {"message":"Unauthorized"}"#;
106        assert!(check_fatal_error(msg).is_some());
107    }
108
109    #[test]
110    fn test_404_is_not_fatal() {
111        let msg = r#"http response status code: 404, response body: {"__type":"ResourceNotFoundException","Message":"Resource not found"}"#;
112        assert!(check_fatal_error(msg).is_none());
113    }
114
115    #[test]
116    fn test_resource_not_found_is_not_fatal() {
117        let msg = r#"Resource of type 'AWS::EC2::VPC' with identifier 'vpc-xxx' was not found"#;
118        assert!(check_fatal_error(msg).is_none());
119    }
120
121    #[test]
122    fn test_400_bad_request_is_not_fatal() {
123        let msg = r#"insert over HTTP error: 400 Bad Request"#;
124        assert!(check_fatal_error(msg).is_none());
125    }
126
127    #[test]
128    fn test_normal_query_error_is_not_fatal() {
129        let msg = r#"query returns error: no such column: foo"#;
130        assert!(check_fatal_error(msg).is_none());

132
133    #[test]

144
