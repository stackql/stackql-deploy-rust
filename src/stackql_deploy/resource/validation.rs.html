3//! # Manifest Validation Module
4//!
5//! Validates a parsed manifest against a set of rules before any command
6//! (build, test, teardown) proceeds.  Each rule is a standalone function
7//! that returns a list of validation errors.  New rules can be added by
8//! implementing a function with the signature
9//! `fn(manifest: &Manifest) -> Vec<ValidationError>` and appending it to
10//! the `RULES` array in [`validate_manifest`].
11
12use std::collections::HashMap;
13
14use crate::resource::manifest::Manifest;
15
16/// A single validation error with a rule name and human-readable message.
17#[derive(Debug, Clone, PartialEq)]
18pub struct ValidationError {
19    /// Machine-readable rule identifier (e.g. `"unique_resource_names"`).
20    pub rule: String,
21    /// Human-readable description of the violation.
22    pub message: String,
23}
24
25impl std::fmt::Display for ValidationError {
26    fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
27        write!(f, "[{}] {}", self.rule, self.message)
28    }
29}
30
31/// Validate a manifest against all registered rules.
32///
33/// Returns `Ok(())` when the manifest is valid, or `Err(Vec<ValidationError>)`
34/// containing every violation found (rules are not short-circuited).
35pub fn validate_manifest(manifest: &Manifest) -> Result<(), Vec<ValidationError>> {
36    // Register rules here.  Each entry is a function that accepts a &Manifest
37    // and returns a Vec<ValidationError>.  Adding a new rule is as simple as
38    // appending another entry to this list.
39    let rules: Vec<fn(&Manifest) -> Vec<ValidationError>> = vec![rule_unique_resource_names];
40
41    let errors: Vec<ValidationError> = rules.iter().flat_map(|rule| rule(manifest)).collect();
42
43    if errors.is_empty() {
44        Ok(())
45    } else {
46        Err(errors)
47    }
48}
49
50// ---------------------------------------------------------------------------
51// Rules
52// ---------------------------------------------------------------------------
53
54/// Resource names within a manifest must be unique.
55///
56/// Because resource-scoped exports use the resource name as a namespace
57/// (e.g. `{{ my_resource.var }}`), duplicate names would create ambiguous
58/// references and silently overwrite immutable scoped exports.
59fn rule_unique_resource_names(manifest: &Manifest) -> Vec<ValidationError> {
60    let mut seen: HashMap<&str, usize> = HashMap::new();
61    let mut errors = Vec::new();
62
63    for (idx, resource) in manifest.resources.iter().enumerate() {
64        if let Some(&first_idx) = seen.get(resource.name.as_str()) {
65            errors.push(ValidationError {
66                rule: "unique_resource_names".to_string(),
67                message: format!(
68                    "Duplicate resource name '{}' at index {} (first seen at index {})",
69                    resource.name, idx, first_idx
70                ),
71            });
72        } else {
73            seen.insert(&resource.name, idx);
74        }
75    }
76
77    errors
78}

95                .iter()

97                    name: n.to_string(),

100                    sql: None,
101                    run: None,
102                    props: vec![],
103                    exports: vec![],
104                    protected: vec![],
105                    description: String::new(),
106                    r#if: None,
107                    skip_validation: None,
108                    skip_on_delete: false,
109                    auth: None,

114        }

196    /// Helper: create a temp stack directory with a manifest and empty resources/.
197    fn write_manifest_file(content: &str) -> tempfile::TempDir {
198        let dir = tempfile::tempdir().unwrap();
199        fs::create_dir_all(dir.path().join("resources")).unwrap();
200        fs::write(dir.path().join("stackql_manifest.yml"), content).unwrap();
201        dir
202    }
