3//! # Common Command Arguments
4//!
5//! This module defines common command-line arguments that can be reused across
6//! different commands in the application.
7
8use clap::{value_parser, Arg, ArgAction};
9use std::str::FromStr;
10
11/// Possible actions to take on failure
12#[derive(Debug, Clone, Copy, PartialEq)]
13pub enum FailureAction {
14    Rollback,
15    Ignore,
16    Error,
17}
18
19impl FromStr for FailureAction {
20    type Err = String;
21
22    fn from_str(s: &str) -> Result<Self, Self::Err> {
23        match s.to_lowercase().as_str() {
24            "rollback" => Ok(FailureAction::Rollback),
25            "ignore" => Ok(FailureAction::Ignore),
26            "error" => Ok(FailureAction::Error),
27            _ => Err(format!("Unknown failure action: {}", s)),
28        }
29    }
30}
31
32// Positional arguments
33/// Common positional argument for the stack directory
34pub fn stack_dir() -> Arg {
35    Arg::new("stack_dir")
36        .required(true)
37        .help("Path to the stack directory containing resources")
38}
39
40/// Common positional argument for the stack environment
41pub fn stack_env() -> Arg {
42    Arg::new("stack_env")
43        .required(true)
44        .help("Environment to deploy to (e.g., `prod`, `dev`, `test`)")
45}
46
47// Optional arguments
48/// Common argument for setting the log level
49pub fn log_level() -> Arg {
50    Arg::new("log-level")
51        .long("log-level")
52        .help("Set the logging level")
53        .default_value("info")
54        .value_parser(clap::builder::PossibleValuesParser::new([
55            "trace", "debug", "info", "warn", "error",
56        ]))
57        .ignore_case(true)
58}
59
60/// Common argument for specifying an environment file
61pub fn env_file() -> Arg {
62    Arg::new("env-file")
63        .long("env-file")
64        .help("Environment variables file")
65        .default_value(".env")
66}
67
68/// Common argument for setting additional environment variables
69pub fn env_var() -> Arg {
70    Arg::new("env")
71        .short('e')
72        .long("env")
73        .help("Set additional environment variables (format: KEY=VALUE)")
74        .action(ArgAction::Append)
75}
76
77/// Common argument for performing a dry run
78pub fn dry_run() -> Arg {
79    Arg::new("dry-run")
80        .long("dry-run")
81        .help("Perform a dry run of the operation")
82        .action(ArgAction::SetTrue)
83}
84
85/// Common argument for showing queries in the output logs
86pub fn show_queries() -> Arg {
87    Arg::new("show-queries")
88        .long("show-queries")
89        .help("Show queries run in the output logs")
90        .action(ArgAction::SetTrue)
91}
92
93/// Common argument for specifying the action on failure
94pub fn on_failure() -> Arg {
95    Arg::new("on-failure")
96        .long("on-failure")
97        .help("Action to take on failure")
98        .value_parser(value_parser!(FailureAction))
99        .default_value("error")
100}
