3//! # Init Command Module
4//!
5//! This module handles the `init` command, which initializes a new StackQL Deploy project structure.
6//! It supports built-in templates for major providers (AWS, Azure, Google) as well as custom templates via URL or file path.
7//!
8//! ## Features
9//! - Initializes project directory structure.
10//! - Supports both embedded templates and custom templates.
11//! - Fetches templates from URLs or uses built-in ones.
12//! - Validates supported providers and applies default providers when necessary.
13//!
14//! ## Example Usage
15//! ```bash
16//! ./stackql-deploy init my-project --provider aws
17//! ./stackql-deploy init my-project --template https://github.com/user/template-repo
18//! ```
19

37enum TemplateSource {
38    Embedded(String), // Built-in template using one of the supported providers
39    Custom(String),   // Custom template path or URL
40}
41
42impl TemplateSource {
43    fn get_sample_res_name(&self) -> &str {
44        match self {
45            TemplateSource::Embedded(provider) => match provider.as_str() {
46                "google" => "example_vpc",
47                "azure" => "example_res_grp",
48                "aws" => "example_vpc",
49                _ => "example_resource",
50            },
51            TemplateSource::Custom(path) => {
52                // Try to determine resource name based on path
53                if path.contains("aws") {
54                    "example_vpc"
55                } else if path.contains("azure") {
56                    "example_res_grp"
57                } else if path.contains("google") {
58                    "example_vpc"
59                } else {
60                    "example_resource"
61                }
62            }
63        }
64    }
65}

67/// Configures the `init` command for the CLI application.
68pub fn command() -> Command {
69    Command::new("init")
70        .about("Initialize a new stackql-deploy project structure")
71        .arg(
72            Arg::new("stack_name")
73                .help("Name of the new stack project")
74                .required(true)
75                .index(1),
76        )
77        .arg(
78            Arg::new("provider")
79                .short('p')
80                .long("provider")
81                .help("Specify a provider (aws, azure, google)")
82                .action(ArgAction::Set)
83                .conflicts_with("template"),
84        )
85        .arg(
86            Arg::new("template")
87                .short('t')
88                .long("template")
89                .help("Template path or URL (e.g., 'aws/starter' or full GitHub URL)")
90                .action(ArgAction::Set)
91                .conflicts_with("provider"),
92        )
93        .arg(
94            Arg::new("env")
95                .short('e')
96                .long("env")
97                .help("Default environment for examples in generated README (default: dev)")
98                .default_value("dev")
99                .action(ArgAction::Set),
100        )
101}

103/// Executes the `init` command to initialize a new project structure.
104pub fn execute(matches: &ArgMatches) {
105    print_unicode_box(
106        "Initializing new project...",
107        crate::utils::display::BorderColor::Yellow,
108    );
109
110    let stack_name = matches
111        .get_one::<String>("stack_name")
112        .expect("Stack name is required");
113
114    let stack_name = stack_name.replace('_', "-").to_lowercase();
115
116    let env = matches
117        .get_one::<String>("env")
118        .expect("Environment defaulted to dev")
119        .to_string();
120
121    // Check if using custom template or provider
122    let template_source = if let Some(template_path) = matches.get_one::<String>("template") {
123        TemplateSource::Custom(template_path.clone())
124    } else {
125        // Get the provider with validation
126        let provider = validate_provider(matches.get_one::<String>("provider").map(|s| s.as_str()));
127        TemplateSource::Embedded(provider)
128    };
129
130    // Create project structure
131    match create_project_structure(&stack_name, &template_source, &env) {
132        Ok(_) => {
133            print_success!("Project '{}' initialized successfully.", stack_name);
134        }
135        Err(e) => {
136            print_error!("Error initializing project: {}", e);
137        }
138    }
139}

141/// Validates the provided provider and returns the appropriate string value.
142fn validate_provider(provider: Option<&str>) -> String {
143    let supported: HashSet<&str> = SUPPORTED_PROVIDERS.iter().cloned().collect();
144
145    match provider {
146        Some(p) if supported.contains(p) => p.to_string(),
147        Some(p) => {
148            print_info!("Provider '{}' is not supported for `init`, supported providers are: {}, defaulting to `{}`",
149                p, SUPPORTED_PROVIDERS.join(", "), DEFAULT_PROVIDER
150            );
151            DEFAULT_PROVIDER.to_string()
152        }
153        _none => {
154            // Silently default to DEFAULT_PROVIDER
155            DEFAULT_PROVIDER.to_string()
156        }
157    }
158}

160/// Fetches template content from a given URL.
161fn fetch_template(url: &str) -> Result<String, String> {
162    let client = Client::builder()
163        .user_agent(concat!("stackql-deploy/", env!("CARGO_PKG_VERSION")))
164        .build()
165        .map_err(|e| format!("Failed to create HTTP client: {}", e))?;
166    let response = client
167        .get(url)
168        .send()
169        .map_err(|e| format!("Failed to fetch template: {}", e))?;
170
171    // Check if response is successful (status code 200-299)
172    if !response.status().is_success() {
173        // Handle 404 and other error status codes
174        if response.status() == StatusCode::NOT_FOUND {
175            return Err(format!("Template not found at URL: {}", url));
176        } else {
177            return Err(format!(
178                "Failed to fetch template: HTTP status {}",
179                response.status()
180            ));
181        }
182    }
183
184    response
185        .text()
186        .map_err(|e| format!("Failed to read template content: {}", e))
187}

189/// Normalizes GitHub URL to raw content URL
190fn normalize_github_url(url: &str) -> String {
191    if url.starts_with("https://github.com") {
192        // Convert github.com URL to raw.githubusercontent.com
193        url.replace("https://github.com", "https://raw.githubusercontent.com")
194            .replace("/tree/", "/")
195    } else {
196        url.to_string()
197    }
198}

200/// Builds full URL or path for templates
201fn build_template_url(template_path: &str, resource_name: &str, file_type: &str) -> String {
202    // Check if template_path is an absolute URL
203    if template_path.starts_with("http://") || template_path.starts_with("https://") {
204        let base_url = normalize_github_url(template_path);
205
206        match file_type {
207            "resource" => format!("{}/resources/{}.iql.template", base_url, resource_name),
208            "manifest" => format!("{}/stackql_manifest.yml.template", base_url),
209            "readme" => format!("{}/README.md.template", base_url),
210            _ => base_url,
211        }
212    } else {
213        // It's a relative path, prepend with GitHub template base
214        let base_url = format!("{}{}", GITHUB_TEMPLATE_BASE, template_path);
215
216        match file_type {
217            "resource" => format!("{}/resources/{}.iql.template", base_url, resource_name),
218            "manifest" => format!("{}/stackql_manifest.yml.template", base_url),
219            "readme" => format!("{}/README.md.template", base_url),
220            _ => base_url,
221        }
222    }
223}

225/// Gets the template content based on the source and type
226fn get_template_content(
227    template_source: &TemplateSource,
228    template_type: &str,
229    resource_name: &str,
230) -> Result<String, String> {
231    match template_source {
232        TemplateSource::Embedded(provider) => {
233            // Use embedded templates
234            match (provider.as_str(), template_type) {
235                ("aws", "resource") => Ok(aws_templates::RESOURCE_TEMPLATE.to_string()),
236                ("aws", "manifest") => Ok(aws_templates::MANIFEST_TEMPLATE.to_string()),
237                ("aws", "readme") => Ok(aws_templates::README_TEMPLATE.to_string()),
238                ("azure", "resource") => Ok(azure_templates::RESOURCE_TEMPLATE.to_string()),
239                ("azure", "manifest") => Ok(azure_templates::MANIFEST_TEMPLATE.to_string()),
240                ("azure", "readme") => Ok(azure_templates::README_TEMPLATE.to_string()),
241                ("google", "resource") => Ok(google_templates::RESOURCE_TEMPLATE.to_string()),
242                ("google", "manifest") => Ok(google_templates::MANIFEST_TEMPLATE.to_string()),
243                ("google", "readme") => Ok(google_templates::README_TEMPLATE.to_string()),
244                _ => Err(format!(
245                    "Unsupported provider or template type: {}, {}",
246                    provider, template_type
247                )),
248            }
249        }
250        TemplateSource::Custom(path) => {
251            // Fetch template from URL or path
252            let template_url = build_template_url(path, resource_name, template_type);
253
254            // Fetch content from URL
255            print_info!("Fetching template from: {}", template_url);
256            fetch_template(&template_url)
257        }
258    }
259}

261/// Creates the project structure for a new StackQL Deploy project.
262fn create_project_structure(
263    stack_name: &str,
264    template_source: &TemplateSource,
265    env: &str,
266) -> Result<(), String> {
267    let cwd =
268        std::env::current_dir().map_err(|e| format!("Failed to get current directory: {}", e))?;
269    let base_path = cwd.join(stack_name);
270
271    // Check if directory already exists
272    if base_path.exists() {
273        return Err(format!("Directory '{}' already exists", stack_name));
274    }
275
276    // Determine sample resource name based on provider
277    let sample_res_name = template_source.get_sample_res_name();
278
279    // Set up template context
280    let mut context = Context::new();
281    context.insert("stack_name", stack_name);
282    context.insert("stack_env", env);
283
284    // First validate that all templates can be fetched before creating any directories
285    let manifest_template = get_template_content(template_source, "manifest", "")?;
286    let readme_template = get_template_content(template_source, "readme", "")?;
287    let resource_template = get_template_content(template_source, "resource", sample_res_name)?;
288
289    // Now create directories
290    let resource_dir = base_path.join("resources");
291    fs::create_dir_all(&resource_dir)
292        .map_err(|e| format!("Failed to create directories: {}", e))?;
293
294    // Create files
295    create_manifest_file(&base_path, &manifest_template, &context)?;
296    create_readme_file(&base_path, &readme_template, &context)?;
297    create_resource_file(&resource_dir, sample_res_name, &resource_template, &context)?;
298
299    Ok(())
300}

302/// Creates a resource file in the specified directory using the provided template and context.
303fn create_resource_file(
304    resource_dir: &Path,
305    sample_res_name: &str,
306    template_str: &str,
307    context: &Context,
308) -> Result<(), String> {
309    // Render template with Tera
310    let resource_content = render_template(template_str, context)
311        .map_err(|e| format!("Template rendering error: {}", e))?;
312
313    let resource_path = resource_dir.join(format!("{}.iql", sample_res_name));
314    let mut file = fs::File::create(resource_path)
315        .map_err(|e| format!("Failed to create resource file: {}", e))?;
316
317    file.write_all(resource_content.as_bytes())
318        .map_err(|e| format!("Failed to write to resource file: {}", e))?;
319
320    Ok(())
321}

363/// Renders a template string using Tera with the provided context.
364fn render_template(template_str: &str, context: &Context) -> Result<String, String> {
365    // Create a one-off Tera instance for rendering a single template
366    let mut tera = Tera::default();
367    tera.add_raw_template("template", template_str)
368        .map_err(|e| format!("Failed to add template: {}", e))?;
369
370    tera.render("template", context)
371        .map_err(|e| format!("Failed to render template: {}", e))
372}
