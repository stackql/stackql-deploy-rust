3//! # Start Server Command Module
4//!
5//! This module provides the `start-server` command for the StackQL Deploy application.
6//! The `start-server` command initializes and starts a local StackQL server based on the
7//! specified configuration options such as mTLS, custom authentication, and logging levels.
8//!
9//! ## Features
10//! - Validates if the server is already running before attempting to start a new instance.
11//! - Supports configuration of mTLS and custom authentication via JSON inputs.
12//! - Allows setting of logging levels for better observability.
13//! - Uses global configuration for host and port.
14//!
15//! ## Example Usage
16//! ```bash
17//! ./stackql-deploy start-server --registry "http://localhost:8000" --log-level INFO
18//! ```
19
20use std::process;
21
22use clap::{Arg, ArgAction, ArgMatches, Command};
23use colored::*;
24
25use crate::app::LOCAL_SERVER_ADDRESSES;
26use crate::globals::{server_host, server_port};
27use crate::utils::display::print_unicode_box;
28use crate::utils::server::{is_server_running, start_server, StartServerOptions};

30/// Configures the `start-server` command for the CLI application.
31pub fn command() -> Command {
32    Command::new("start-server")
33        .about("Start the stackql server")
34        .arg(
35            Arg::new("registry")
36                .short('r')
37                .long("registry")
38                .help("[OPTIONAL] Custom registry URL")
39                .action(ArgAction::Set),
40        )
41        .arg(
42            Arg::new("mtls_config")
43                .short('m')
44                .long("mtls-config")
45                .help("[OPTIONAL] mTLS configuration for the server (JSON object)")
46                .action(ArgAction::Set),
47        )
48        .arg(
49            Arg::new("custom_auth_config")
50                .short('a')
51                .long("custom-auth-config")
52                .help("[OPTIONAL] Custom provider authentication configuration for the server (JSON object)")
53                .action(ArgAction::Set),
54        )
55        .arg(
56            Arg::new("log_level")
57                .short('l')
58                .long("log-level")
59                .help("[OPTIONAL] Server log level (default: WARN)")
60                .value_parser(["TRACE", "DEBUG", "INFO", "WARN", "ERROR", "FATAL"])
61                .action(ArgAction::Set),
62        )
63}
64
65/// Executes the `start-server` command.
66pub fn execute(matches: &ArgMatches) {
67    print_unicode_box(
68        "Starting stackql server...",
69        crate::utils::display::BorderColor::Yellow,
70    );
71
72    // Use global vars for host and port
73    let port = server_port();
74    let host = server_host().to_string();
75
76    // Validate host - must be localhost or 0.0.0.0
77    if !LOCAL_SERVER_ADDRESSES.contains(&host.as_str()) {
78        eprintln!(
79            "{}",
80            "Error: Host must be 'localhost' or '0.0.0.0' for local server setup.".red()
81        );
82        eprintln!("The start-server command is only for starting a local server instance.");
83        process::exit(1);
84    }
85
86    // Check if server is already running
87    if is_server_running(port) {
88        println!(
89            "{}",
90            format!(
91                "Server is already running on port {}. No action needed.",
92                port
93            )
94            .yellow()
95        );
96        process::exit(0);
97    }
98
99    // Get optional settings
100    let registry = matches.get_one::<String>("registry").cloned();
101    let mtls_config = matches.get_one::<String>("mtls_config").cloned();
102    let custom_auth_config = matches.get_one::<String>("custom_auth_config").cloned();
103    let log_level = matches.get_one::<String>("log_level").cloned();
104
105    // Create server options
106    let options = StartServerOptions {
107        host: host.clone(),
108        port,
109        registry,
110        mtls_config,
111        custom_auth_config,
112        log_level,
113    };
114
115    // Start the server
116    match start_server(&options) {
117        Ok(_pid) => {
118            println!(
119                "{}",
120                format!("Server is listening on {}:{}", options.host, options.port).green()
121            );
122        }
123        Err(e) => {
124            eprintln!("{}", format!("Failed to start server: {}", e).red());
125            process::exit(1);
126        }
127    }
128}
